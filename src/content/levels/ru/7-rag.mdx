        ```python
        import re
        import sys
        from pathlib import Path

        from sqlalchemy import delete, func, select

        from app.db import SessionLocal
        from app.models import Chunk, Document
        from app.services.embeddings import embed_passages

        MAX_CHARS = 800
        OVERLAP_SENTENCES = 1


        def split_sections(text: str) -> list[tuple[str, str]]:
            """Cut a markdown file at its headings: [(heading, text under it), ...]."""
            sections, heading, lines = [], "", []
            for line in text.splitlines():
                if line.startswith("#"):
                    if "".join(lines).strip():
                        sections.append((heading, "\n".join(lines).strip()))
                    heading, lines = line.lstrip("#").strip(), []
                else:
                    lines.append(line)
            if "".join(lines).strip():
                sections.append((heading, "\n".join(lines).strip()))
            return sections


        def split_sentences(text: str) -> list[str]:
            return [s for s in re.split(r"(?<=[.!?])\s+", text) if s]


        def chunk_section(text: str) -> list[str]:
            """Pack whole sentences into chunks of at most MAX_CHARS; a new chunk repeats the last sentences."""
            chunks, current = [], []
            for sentence in split_sentences(text):
                if current and len(" ".join(current + [sentence])) > MAX_CHARS:
                    chunks.append(" ".join(current))
                    current = current[len(current) - OVERLAP_SENTENCES :]
                current.append(sentence)
            if current:
                chunks.append(" ".join(current))
            return chunks


        def ingest_file(session, path: Path) -> int:
            source = path.name
            # Ingesting a file again replaces it: remove its old chunks first.
            old = select(Document.id).where(Document.source == source)
            session.execute(delete(Chunk).where(Chunk.document_id.in_(old)))
            session.execute(delete(Document).where(Document.source == source))

            sections = split_sections(path.read_text(encoding="utf-8"))
            document = Document(title=sections[0][0] or path.stem, source=source)
            session.add(document)
            session.flush()

            # The heading goes into every chunk, so each chunk makes sense alone.
            pieces = [(heading, f"{heading}\n{text}") for heading, body in sections for text in chunk_section(body)]
            vectors = embed_passages([content for _, content in pieces])
            session.add_all(
                Chunk(document_id=document.id, section=heading, content=content, embedding=vector)
                for (heading, content), vector in zip(pieces, vectors)
            )
            return len(pieces)


        def main(folder: str) -> None:
            with SessionLocal() as session:
                for path in sorted(Path(folder).glob("*.md")):
                    print(f"{path.name}: {ingest_file(session, path)} chunks")
                session.commit()
                sample = session.scalars(select(Chunk).order_by(func.random()).limit(5)).all()
                for chunk in sample:
                    print(f"\n--- chunk {chunk.id}, section: {chunk.section}\n{chunk.content}")


        if __name__ == "__main__":
            main(sys.argv[1])
        ```

        ```python
        KEYWORD_SQL = text("""
            SELECT chunks.id, chunks.content, chunks.section, documents.source
            FROM chunks
            JOIN documents ON documents.id = chunks.document_id,
                 websearch_to_tsquery('simple', :words) AS q
            WHERE chunks.tsv @@ q
            ORDER BY ts_rank(chunks.tsv, q) DESC
            LIMIT :k
        """)


        def keyword_search(session, question: str, k: int) -> list[Found]:
            # Any of the words may match, not all of them: "a or b or c".
            words = " or ".join(re.findall(r"[\w-]+", question))
            rows = session.execute(KEYWORD_SQL, {"words": words, "k": k}).all()
            return [Found(r.id, r.content, r.source, r.section) for r in rows]


        def rrf(*lists: list[Found], k: int = 60) -> list[Found]:
            """Reciprocal rank fusion: each chunk scores 1 / (k + rank) in every list it is in."""
            scores: dict[int, float] = {}
            by_id: dict[int, Found] = {}
            for results in lists:
                for rank, found in enumerate(results, start=1):
                    scores[found.id] = scores.get(found.id, 0) + 1 / (k + rank)
                    by_id[found.id] = found
            return [by_id[i] for i in sorted(scores, key=scores.get, reverse=True)]


        def hybrid_search(session, question: str, k: int = 20) -> list[Found]:
            return rrf(vector_search(session, question, k), keyword_search(session, question, k))
        ```

        ```python
        RERANK_MODEL = "cross-encoder/mmarco-mMiniLMv2-L12-H384-v1"
        CANDIDATES = 20


        @lru_cache
        def get_reranker():
            from sentence_transformers import CrossEncoder

            return CrossEncoder(RERANK_MODEL)


        def rerank(question: str, found: list[Found]) -> list[Found]:
            """The cross-encoder reads the question and each chunk together; best first."""
            if not found:
                return []
            scores = get_reranker().predict([(question, f.content) for f in found])
            return [f for f, _ in sorted(zip(found, scores), key=lambda pair: pair[1], reverse=True)]


        def retrieve(session, question: str) -> list[Found]:
            return rerank(question, hybrid_search(session, question, CANDIDATES))
        ```

            hits = reciprocal = faithful = correct = 0.0

            for item in golden:

                    ranked = retrieve(session, item["question"])
                if item["contains"]:
                    answerable += 1
                    rank = next((i for i, c in enumerate(ranked, start=1) if item["contains"] in c.content), None)
                    if rank is not None:
                        hits += rank <= 5
                        reciprocal += 1 / rank
                chunks = ranked[:KEEP]

                verdict = judge(item["question"], item["answer"], chunks, answer)
                faithful += verdict.faithful
                correct += verdict.correct
                print(f"{'ok ' if verdict.correct else 'BAD'} {item['question']}\n    {answer}")

            n = len(golden)
            numbers = [hits / answerable, reciprocal / answerable, faithful / n, correct / n]
            print("hit@5 {:.2f}  MRR {:.2f}  faithfulness {:.2f}  correctness {:.2f}".format(*numbers))

## Два пайплайна

RAG-система — это два пайплайна, которые встречаются в базе данных.
**Загрузка** работает один раз на документ: разобрать его, почистить,
разрезать на чанки, превратить каждый чанк в эмбеддинг и сохранить вместе
с источником. **Запрос** работает на каждый вопрос: превратить вопрос в
эмбеддинг, найти чанки-кандидаты, переранжировать их, собрать промпт,
сгенерировать ответ и сослаться на источники.

## Разбор: здесь качество выигрывается или проигрывается

Прежде чем что-то резать на чанки, программа должна прочитать твои
документы, — а читают программы плохо. PDF теряют таблицы, один и тот же
колонтитул повторяется на каждой странице, а отсканированная страница —
это картинка без текста, пока её не прочитает OCR. **Всегда смотри на
разобранный текст своими глазами** — хотя бы на несколько страниц каждого
вида документов.

По ходу сохраняй метаданные: исходный файл, страницу, заголовок раздела.
Из них получатся ссылки на источники, и по ним ты сможешь удалить чанки
документа, когда документ поменяется.

## Нарезка на чанки

Чанк должен быть **одной мыслью, понятной самой по себе**. Слишком
маленький — и факт разрезан пополам. Слишком большой — и чанк сразу про
пять вещей, поэтому немного подходит к любому вопросу и ни к одному —
хорошо.

Начни с нескольких сотен токенов и перекрытия 10–20%. Режь там, где у
текста уже есть швы, — по заголовкам, абзацам, предложениям, — а не
вслепую по числу символов. Заголовок раздела перед каждым чанком помогает
чанку быть понятным самому по себе.

## Эмбеддинги

Документы и вопросы должна встраивать одна и та же модель эмбеддингов,
иначе они окажутся на разных картах. Некоторые модели ждут префикс: e5
хочет `passage: ` перед документами и `query: ` перед вопросами. А новая
модель — значит, заново встроить всё; почему — ты знаешь с уровня 5.

## Гибридный поиск

Векторный поиск находит **смысл**: «вернуть деньги» находит правило о
возвратах. Поиск по словам находит **точные вещи**: номера заказов, имена,
коды. Для эмбеддинга `KZ-4471` и `KZ-6642` выглядят почти одинаково, а для
поиска по словам это разные слова. Используй оба.

В Postgres поиск по словам встроен:

```sql
ALTER TABLE chunks ADD COLUMN tsv tsvector
  GENERATED ALWAYS AS (to_tsvector('simple', content)) STORED;
CREATE INDEX ON chunks USING gin (tsv);

SELECT id, ts_rank(tsv, q) AS score
FROM chunks, websearch_to_tsquery('simple', 'возврат or чек') q
WHERE tsv @@ q ORDER BY score DESC LIMIT 20;
```

Потом слей два списка через **reciprocal rank fusion** (слияние по
рангам): чанк получает `1 / (60 + место)` в каждом списке, где он есть, и
баллы складываются. Именно места, а не сырые оценки, — поэтому двум
поисковикам не нужно договариваться, что такое «оценка». Чанк, который
нравится обоим, поднимается наверх.

## Реранкинг

Поиск быстрый, потому что сравнивает вопрос с каждым чанком по
отдельности. **Кросс-энкодер-реранкер** читает вопрос и чанк вместе — это
медленнее и гораздо точнее. Поэтому: искать широко (20–30 кандидатов), а
потом пусть реранкер оставит лучшие 3–5.

## Промпт

Скажи модели отвечать только по пронумерованным источникам, ссылаться на
них так: `[1]`, — и, главное, честно говорить, когда в источниках ответа
нет. Возвращай источники тому, кто спрашивал, чтобы человек мог проверить
ссылку.

## Оценка: то, что отличает инженерию от гадания

Собери **эталонный набор (golden set)**: 30 или больше настоящих
вопросов, у каждого — чанк (или документ и страница), где лежит ответ, и
короткий эталонный ответ. Добавь несколько вопросов, на которые в
документах ответа нет. Потом измеряй две половины по отдельности:

- **Поиск.** **Hit rate@k**: в какой доле вопросов нужный чанк попал в
  первые k? **MRR**: насколько высоко он в среднем? (1 за первое место, ½
  за второе, ⅓ за третье.)
- **Ответы.** **Достоверность (faithfulness)**: подтверждается ли каждое
  утверждение найденными чанками? **Правильность (correctness)**: означает
  ли ответ то же, что эталонный? Обе оценивает LLM по неизменному
  промпту — а часть её вердиктов ты проверяешь своими глазами.

Запускай оценку после каждого изменения и веди таблицу результатов.

## Когда ответ неверный

Сначала посмотри на найденные чанки. **Нужного чанка нет?** Это проблема
поиска: разбор, нарезка или сам поиск. **Нужный чанк есть?** Это проблема
генерации: промпт, слишком много контекста, модель. Одно это правило
экономит дни, потраченные на подкручивание не того.

## В продакшене

**Изменённые документы.** Перед повторной загрузкой документа удали его
старые чанки по id документа — иначе отвечать будут и старая, и новая
версия.

**Права доступа.** Фильтруй по метаданным прямо в запросе. Никогда не
доставай то, что пользователю видеть нельзя: чанк в промпте — это чанк,
который модель может пересказать.

**Кэш.** Вопросы повторяются. Кэшируй эмбеддинги частых вопросов, а там,
где документы меняются редко, — и целые ответы (стикеры из уровня 5).

**Бюджет задержки.** Дай каждому этапу свой бюджет — эмбеддинг, поиск,
реранкинг, генерация — и пиши время каждого в лог. Когда ответы станут
медленными, ты будешь знать, какой этап виноват, а не гадать.

## Что запомнить

1. **Ответ неверный? Сначала прочитай найденные чанки.** Большинство галлюцинаций — это провалы поиска.
2. **Чанк — одна мысль, понятная сама по себе.**
3. **Векторы — для смысла, слова — для точных вещей, реранкер — чтобы выбрать.**
4. **«Этого нет в документах» — это фича.**
5. **Поменяй одну вещь, прогони эталонный набор, оставь или откати.**
