        ```python
        import logging
        import time
        from pathlib import Path

        from app.config import settings
        from app.llm import client

        log = logging.getLogger("quest-api")
        PROMPTS = Path(__file__).parents[2] / "prompts"
        ANSWER_PROMPT = "answer_v1"


        def load_prompt(name: str) -> str:
            return (PROMPTS / f"{name}.txt").read_text()


        def llm_answer(question: str) -> str:
            start = time.perf_counter()
            response = client.chat.completions.create(
                model=settings.llm_model,
                messages=[
                    {"role": "system", "content": load_prompt(ANSWER_PROMPT)},
                    {"role": "user", "content": question},
                ],
                temperature=0.2,
                max_tokens=300,
            )
            ms = (time.perf_counter() - start) * 1000
            choice = response.choices[0]
            log.info(
                "llm model=%s prompt=%s in=%d out=%d %.0fms finish=%s",
                settings.llm_model, ANSWER_PROMPT, response.usage.prompt_tokens,
                response.usage.completion_tokens, ms, choice.finish_reason,
            )
            return choice.message.content
        ```

      - |
        Add a streaming version to `app/services/answer.py` (with `from collections.abc import Iterator` at the top):
        ```python
        def llm_stream(question: str) -> Iterator[str]:
            start = time.perf_counter()
            first_token_ms = None
            out_tokens = 0
            stream = client.chat.completions.create(
                model=settings.llm_model,
                messages=[
                    {"role": "system", "content": load_prompt(ANSWER_PROMPT)},
                    {"role": "user", "content": question},
                ],
                temperature=0.2,
                max_tokens=300,
                stream=True,
                stream_options={"include_usage": True},
            )
            for chunk in stream:
                if chunk.choices and chunk.choices[0].delta.content:
                    if first_token_ms is None:
                        first_token_ms = (time.perf_counter() - start) * 1000
                    yield chunk.choices[0].delta.content
                if chunk.usage:
                    out_tokens = chunk.usage.completion_tokens
            total_ms = (time.perf_counter() - start) * 1000
            log.info(
                "llm stream model=%s out=%d first_token=%.0fms total=%.0fms",
                settings.llm_model, out_tokens, first_token_ms or 0, total_ms,
            )
        ```
        `include_usage` asks the server for one last chunk with the token counts.

        Add the extraction to `app/services/answer.py` (with `from pydantic import ValidationError` and `from app.schemas import Complaint` at the top). The schema goes along with the request, so the server can hold the model to it — and Pydantic checks anyway:
        ```python
        EXTRACT_PROMPT = "extract_v1"


        class ExtractionFailed(Exception):
            pass


        def extract_complaint(text: str) -> Complaint:
            schema = {"name": "complaint", "schema": Complaint.model_json_schema()}
            for attempt in (1, 2):
                response = client.chat.completions.create(
                    model=settings.llm_model,
                    messages=[
                        {"role": "system", "content": load_prompt(EXTRACT_PROMPT)},
                        {"role": "user", "content": text},
                    ],
                    temperature=0,
                    max_tokens=300,
                    response_format={"type": "json_schema", "json_schema": schema},
                )
                raw = response.choices[0].message.content
                try:
                    return Complaint.model_validate_json(raw)
                except ValidationError as error:
                    log.warning("extract attempt %d failed: %s", attempt, error.errors()[0]["msg"])
            raise ExtractionFailed("the model did not return a valid complaint")
        ```
      - |
        Add the route (import `HTTPException`, the two schemas, `extract_complaint` and `ExtractionFailed`). A broken answer from the model is not the caller's fault, so it gets `502`, not `422`:
        ```python
        @router.post("/extract", response_model=Complaint)
        def extract(body: ExtractRequest) -> Complaint:
            try:
                return extract_complaint(body.text)
            except ExtractionFailed as error:
                raise HTTPException(status_code=502, detail=str(error))
        ```

      - |
        Write `scripts/load_test.py`. Twenty questions, at most four at a time — the `Semaphore` is the bouncer at the door:
        ```python
        import asyncio
        import time

        from openai import AsyncOpenAI

        from app.config import settings

        client = AsyncOpenAI(base_url=settings.llm_base_url, api_key=settings.llm_api_key, timeout=30, max_retries=2)
        limit = asyncio.Semaphore(4)
        in_flight = 0


        async def ask(number: int) -> float:
            global in_flight
            async with limit:
                in_flight += 1
                print(f"start {number:2}  in flight: {in_flight}")
                start = time.perf_counter()
                await client.chat.completions.create(
                    model=settings.llm_model,
                    messages=[{"role": "user", "content": f"Tell me fun fact number {number} about otters, in one sentence."}],
                    max_tokens=80,
                )
                seconds = time.perf_counter() - start
                in_flight -= 1
                print(f"done  {number:2}  {seconds:.1f}s")
                return seconds


        async def main() -> None:
            start = time.perf_counter()
            durations = await asyncio.gather(*(ask(n) for n in range(1, 21)))
            print(f"20 calls in {time.perf_counter() - start:.1f}s, slowest {max(durations):.1f}s")


        asyncio.run(main())
        ```

Until now `/ask` has been pretending. In this level it asks a real
language model — and you learn to treat that model like any other service
over a network: with limits, timeouts, logs and a healthy distrust of
what comes back.

**Where do you get a model?** Ask your mentor which endpoint to use. If
there is none, run a small model on your laptop with Ollama, which serves
an OpenAI-compatible API at `http://localhost:11434/v1`. Mission 1 shows

## The shape of the API

Most model servers — vLLM, Ollama and many others — speak the same
language as OpenAI's API. So one client works with all of them; you only
change the address:

```python
from openai import OpenAI

client = OpenAI(base_url=settings.llm_base_url, api_key=settings.llm_api_key)
response = client.chat.completions.create(
    model=settings.llm_model,
    messages=[
        {"role": "system", "content": "You answer in one short sentence."},
        {"role": "user", "content": "What is a port?"},
    ],
    temperature=0.2,
    max_tokens=200,
)
print(response.choices[0].message.content, response.usage)
```

The **system** message sets the rules; the **user** message is the
question. `GET /v1/models` lists the exact model names the server
accepts — copy them, do not guess them.

## Tokens and the context window

Models read and write **tokens**: pieces of words. "Otter" may be one
token; a long rare word may be four. Everything counts against one limit,
the **context window**: the system prompt, any documents, the question —
and the answer still to be written. Prompt tokens plus `max_tokens` must
fit.

<Diagram name="context-window" />

Every response tells you what it cost in `usage`: `prompt_tokens` went
in, `completion_tokens` came out. And every response says why it stopped,
in `finish_reason`: `stop` means the model finished; `length` means it
ran out of `max_tokens` mid-thought.

What happens when it does not fit depends on the server. vLLM refuses
with a `400`. Ollama quietly cuts off the start of your prompt and answers
anyway — which is worse, because nothing tells you. Reading `usage` is how
you find out.

## Streaming

A model writes one token at a time. Without streaming, you wait for the
last one before you see the first. With streaming, the server sends each
piece as it is made, as **server-sent events**: lines that start with
`data:`.

Two numbers describe the speed: **time to first token** — how long the
user stares at nothing — and **tokens per second** after that. In FastAPI,
a `StreamingResponse` passes the pieces on to the caller as they come.

## Things fail

A model server is a service over a network, and it fails like one: it
times out, it answers `429 Too Many Requests` when you push too hard, it
gives `5xx` while it restarts. Give every client a `timeout=` and a
`max_retries=`. The client waits a little longer before each new try —
and stops after the limit. **Never retry forever**: a stuck server
plus endless retries is a traffic jam you built yourself.

## Structured output

When code reads the answer, prose is not enough. Ask for JSON that
matches a schema, pass the schema along (`response_format`) so the server
can hold the model to it, and validate the result with Pydantic anyway.
Then decide what happens when it does not validate: retry once, then fail
clearly. For extraction, use `temperature=0`: you want the same form
filled the same way, not creativity.

## Prompts are code

A prompt changes what your service does, the same way code does. So it lives in a
file, in git, with a version in its name: `answer_v1.txt`,
`answer_v2.txt`. Change one thing at a time, and compare the results on
the same questions. "I tweaked the prompt and now it feels better" is not
a result.

## Many requests at once

Waiting for a model is mostly waiting — the work you met in Level 3's
event loop. With the async client, one process can wait for many answers
at once. But the server has limits too, so you put a bouncer at the door:
an `asyncio.Semaphore(4)` lets at most four calls in at a time.

```python
limit = asyncio.Semaphore(4)

async def ask(question: str) -> str:
    async with limit:
        response = await client.chat.completions.create(...)
        return response.choices[0].message.content
```

## Where the model lives

A model server such as vLLM loads the model onto a GPU once and serves
many callers: `vllm serve <model> --port 8000`. The model's weights fill
most of the GPU's memory before the first question arrives; `nvidia-smi`
shows how much. You will mostly be a caller — but now you know what sits
on the other side of the port.

## What to remember

1. **Prompt plus `max_tokens` must fit the window.** Read `usage`.
2. **Check `finish_reason`.** `length` means unfinished.
3. **Every call has a timeout and a retry limit.** Never retry forever.
4. **Never trust model JSON without validating it.**
5. **Latency is two numbers:** time to first token, and tokens per second.
