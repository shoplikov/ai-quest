        Write `app/ingest.py`. It cuts each file at its headings, packs whole sentences into chunks, and puts the heading in front of every chunk, so each one makes sense alone:
        ```python
        import re
        import sys
        from pathlib import Path

        from sqlalchemy import delete, func, select

        from app.db import SessionLocal
        from app.models import Chunk, Document
        from app.services.embeddings import embed_passages

        MAX_CHARS = 800
        OVERLAP_SENTENCES = 1


        def split_sections(text: str) -> list[tuple[str, str]]:
            """Cut a markdown file at its headings: [(heading, text under it), ...]."""
            sections, heading, lines = [], "", []
            for line in text.splitlines():
                if line.startswith("#"):
                    if "".join(lines).strip():
                        sections.append((heading, "\n".join(lines).strip()))
                    heading, lines = line.lstrip("#").strip(), []
                else:
                    lines.append(line)
            if "".join(lines).strip():
                sections.append((heading, "\n".join(lines).strip()))
            return sections


        def split_sentences(text: str) -> list[str]:
            return [s for s in re.split(r"(?<=[.!?])\s+", text) if s]


        def chunk_section(text: str) -> list[str]:
            """Pack whole sentences into chunks of at most MAX_CHARS; a new chunk repeats the last sentences."""
            chunks, current = [], []
            for sentence in split_sentences(text):
                if current and len(" ".join(current + [sentence])) > MAX_CHARS:
                    chunks.append(" ".join(current))
                    current = current[len(current) - OVERLAP_SENTENCES :]
                current.append(sentence)
            if current:
                chunks.append(" ".join(current))
            return chunks


        def ingest_file(session, path: Path) -> int:
            source = path.name
            # Ingesting a file again replaces it: remove its old chunks first.
            old = select(Document.id).where(Document.source == source)
            session.execute(delete(Chunk).where(Chunk.document_id.in_(old)))
            session.execute(delete(Document).where(Document.source == source))

            sections = split_sections(path.read_text(encoding="utf-8"))
            document = Document(title=sections[0][0] or path.stem, source=source)
            session.add(document)
            session.flush()

            # The heading goes into every chunk, so each chunk makes sense alone.
            pieces = [(heading, f"{heading}\n{text}") for heading, body in sections for text in chunk_section(body)]
            vectors = embed_passages([content for _, content in pieces])
            session.add_all(
                Chunk(document_id=document.id, section=heading, content=content, embedding=vector)
                for (heading, content), vector in zip(pieces, vectors)
            )
            return len(pieces)


        def main(folder: str) -> None:
            with SessionLocal() as session:
                for path in sorted(Path(folder).glob("*.md")):
                    print(f"{path.name}: {ingest_file(session, path)} chunks")
                session.commit()
                sample = session.scalars(select(Chunk).order_by(func.random()).limit(5)).all()
                for chunk in sample:
                    print(f"\n--- chunk {chunk.id}, section: {chunk.section}\n{chunk.content}")


        if __name__ == "__main__":
            main(sys.argv[1])
        ```

        ```python
        import re
        from dataclasses import dataclass
        from functools import lru_cache

        from sqlalchemy import select, text

        from app.models import Chunk, Document
        from app.services.embeddings import embed_query


        @dataclass
        class Found:
            id: int
            content: str
            document: str
            section: str


        def vector_search(session, question: str, k: int) -> list[Found]:
            distance = Chunk.embedding.cosine_distance(embed_query(question))
            rows = session.execute(
                select(Chunk.id, Chunk.content, Chunk.section, Document.source)
                .join(Document, Document.id == Chunk.document_id)
                .order_by(distance)
                .limit(k)
            ).all()
            return [Found(r.id, r.content, r.source, r.section) for r in rows]
        ```

        RAG_PROMPT = "rag_v1"


        def build_prompt(question: str, chunks: list[Found]) -> str:
            sources = "\n\n".join(f"[{i}] {chunk.content}" for i, chunk in enumerate(chunks, start=1))
            return f"Sources:\n{sources}\n\nQuestion: {question}"


        def answer_from(question: str, chunks: list[Found]) -> str:
            start = time.perf_counter()
            response = client.chat.completions.create(
                model=settings.llm_model,
                messages=[
                    {"role": "system", "content": load_prompt(RAG_PROMPT)},
                    {"role": "user", "content": build_prompt(question, chunks)},
                ],
                temperature=0,
                max_tokens=300,
            )
            ms = (time.perf_counter() - start) * 1000
            log.info(
                "llm model=%s prompt=%s in=%d out=%d %.0fms chunks=%s",
                settings.llm_model, RAG_PROMPT, response.usage.prompt_tokens,
                response.usage.completion_tokens, ms, [c.id for c in chunks],
            )
            return response.choices[0].message.content


        def rag_answer(question: str) -> tuple[str, list[Found]]:
            with SessionLocal() as session:
                chunks = vector_search(session, question, 5)
            return answer_from(question, chunks), chunks

      - |
        Add keyword search and the merge to `app/services/retrieval.py`:
        ```python
        KEYWORD_SQL = text("""
            SELECT chunks.id, chunks.content, chunks.section, documents.source
            FROM chunks
            JOIN documents ON documents.id = chunks.document_id,
                 websearch_to_tsquery('simple', :words) AS q
            WHERE chunks.tsv @@ q
            ORDER BY ts_rank(chunks.tsv, q) DESC
            LIMIT :k
        """)


        def keyword_search(session, question: str, k: int) -> list[Found]:
            # Any of the words may match, not all of them: "a or b or c".
            words = " or ".join(re.findall(r"[\w-]+", question))
            rows = session.execute(KEYWORD_SQL, {"words": words, "k": k}).all()
            return [Found(r.id, r.content, r.source, r.section) for r in rows]


        def rrf(*lists: list[Found], k: int = 60) -> list[Found]:
            """Reciprocal rank fusion: each chunk scores 1 / (k + rank) in every list it is in."""
            scores: dict[int, float] = {}
            by_id: dict[int, Found] = {}
            for results in lists:
                for rank, found in enumerate(results, start=1):
                    scores[found.id] = scores.get(found.id, 0) + 1 / (k + rank)
                    by_id[found.id] = found
            return [by_id[i] for i in sorted(scores, key=scores.get, reverse=True)]


        def hybrid_search(session, question: str, k: int = 20) -> list[Found]:
            return rrf(vector_search(session, question, k), keyword_search(session, question, k))
        ```
        Why the `or`: `websearch_to_tsquery` alone joins the words with AND, so a whole question — "What is order KZ-4471?" — must match every word, and finds nothing.

    title: "Second opinion"
    why: "A reranker reads every candidate together with the question and keeps the few that really answer it — and the model learns to say \"I don't know\"."
    steps:
      - |
        Add the reranker to `app/services/retrieval.py`. A cross-encoder is slower than vector search but much sharper, so it only reads the 20 candidates that search found. This one is multilingual and runs on CPU; it is downloaded on first use into your `hf-cache` volume:
        ```python
        RERANK_MODEL = "cross-encoder/mmarco-mMiniLMv2-L12-H384-v1"
        CANDIDATES = 20


        @lru_cache
        def get_reranker():
            from sentence_transformers import CrossEncoder

            return CrossEncoder(RERANK_MODEL)


        def rerank(question: str, found: list[Found]) -> list[Found]:
            """The cross-encoder reads the question and each chunk together; best first."""
            if not found:
                return []
            scores = get_reranker().predict([(question, f.content) for f in found])
            return [f for f, _ in sorted(zip(found, scores), key=lambda pair: pair[1], reverse=True)]


        def retrieve(session, question: str) -> list[Found]:
            return rerank(question, hybrid_search(session, question, CANDIDATES))
        ```

        The judge gets a fixed prompt, kept in a file like any prompt — `eval/judge_v1.txt`:
        ```
        You check one answer from a question-answering system. Return JSON with two fields.
        faithful: true if every fact in the answer is supported by the sources. An answer that says the information is not in the documents is faithful.
        correct: true if the answer means the same as the reference answer. Extra details are fine; missing or different facts are not.
        ```
      - |
        Write `eval/run.py` (and an empty `eval/__init__.py`):
        ```python
        import json
        import sys
        from datetime import date
        from pathlib import Path

        from pydantic import BaseModel

        from app.config import settings
        from app.db import SessionLocal
        from app.llm import client
        from app.services.answer import KEEP, answer_from, build_prompt
        from app.services.retrieval import retrieve

        HERE = Path(__file__).parent
        HEADER = "| date | change | hit@5 | MRR | faithfulness | correctness |\n|---|---|---|---|---|---|\n"


        class Verdict(BaseModel):
            faithful: bool
            correct: bool


        def judge(question: str, reference: str, chunks, answer: str) -> Verdict:
            task = f"{build_prompt(question, chunks)}\n\nReference answer: {reference}\n\nAnswer to check: {answer}"
            response = client.chat.completions.create(
                model=settings.llm_model,
                messages=[
                    {"role": "system", "content": (HERE / "judge_v1.txt").read_text()},
                    {"role": "user", "content": task},
                ],
                temperature=0,
                max_tokens=100,
                response_format={"type": "json_schema", "json_schema": {"name": "verdict", "schema": Verdict.model_json_schema()}},
            )
            return Verdict.model_validate_json(response.choices[0].message.content)


        def main(change: str) -> None:
            golden = [json.loads(line) for line in (HERE / "golden.jsonl").read_text().splitlines() if line.strip()]
            hits = reciprocal = faithful = correct = 0.0
            answerable = 0
            for item in golden:
                with SessionLocal() as session:
                    ranked = retrieve(session, item["question"])
                if item["contains"]:
                    answerable += 1
                    rank = next((i for i, c in enumerate(ranked, start=1) if item["contains"] in c.content), None)
                    if rank is not None:
                        hits += rank <= 5
                        reciprocal += 1 / rank
                chunks = ranked[:KEEP]
                answer = answer_from(item["question"], chunks)
                verdict = judge(item["question"], item["answer"], chunks, answer)
                faithful += verdict.faithful
                correct += verdict.correct
                print(f"{'ok ' if verdict.correct else 'BAD'} {item['question']}\n    {answer}")

            n = len(golden)
            numbers = [hits / answerable, reciprocal / answerable, faithful / n, correct / n]
            print("hit@5 {:.2f}  MRR {:.2f}  faithfulness {:.2f}  correctness {:.2f}".format(*numbers))
            results = HERE / "results.md"
            if not results.exists():
                results.write_text(HEADER)
            with results.open("a") as f:
                f.write(f"| {date.today()} | {change} | " + " | ".join(f"{x:.2f}" for x in numbers) + " |\n")


        if __name__ == "__main__":
            main(sys.argv[1] if len(sys.argv) > 1 else "no description")
        ```

In Level 6 the model answered from memory, and when it did not know the
café's opening hours, it made some up. In this level it gets the handbook.
**RAG** — retrieval-augmented generation — means: find the right pages
first, then answer from them, and say which pages you used. Building one
takes an afternoon. Making it trustworthy takes the three weeks of this
level.

## Two pipelines

A RAG system is two pipelines that meet in the database. **Ingest** runs
once per document: parse it, clean it, cut it into chunks, embed every
chunk and store it with its source. **Query** runs for every question:
embed the question, retrieve candidate chunks, rerank them, build the
prompt, generate the answer and cite the sources.

## Parsing: where quality is won or lost

Before anything is chunked, a program has to read your documents — and
programs read badly. PDFs lose their tables, the same header repeats on
every page, a scanned page is a picture with no text at all until OCR
reads it. **Always look at the parsed text with your own eyes**, at least
for a few pages of each kind of document.

Keep the metadata as you go: the source file, the page, the section
title. They become your citations, and they let you delete a document's
chunks when the document changes.

## Chunking

A chunk should be **one idea that makes sense alone**. Too small, and a
fact gets cut in half. Too big, and the chunk is about five things at
once, so it matches every question a little and none of them well.

Start with a few hundred tokens and 10–20% overlap. Prefer cutting where
the text already has seams — headings, paragraphs, sentences — over
counting characters blindly. Putting the section heading in front of each
chunk helps it make sense alone.

## Embeddings

The same embedding model has to embed both the documents and the
questions, or they land on different maps. Some models expect a prefix:
e5 wants `passage: ` in front of documents and `query: ` in front of
questions. And a new model means re-embedding everything — Level 5 told
you why.

## Hybrid search

Vector search finds **meaning**: "money back" finds the refund rule.
Keyword search finds **exact things**: order numbers, names, codes. To an
embedding, `KZ-4471` and `KZ-6642` look almost alike; to keyword search
they are different words. Use both.

Postgres has keyword search built in:

```sql
ALTER TABLE chunks ADD COLUMN tsv tsvector
  GENERATED ALWAYS AS (to_tsvector('simple', content)) STORED;
CREATE INDEX ON chunks USING gin (tsv);

SELECT id, ts_rank(tsv, q) AS score
FROM chunks, websearch_to_tsquery('simple', 'refund or receipt') q
WHERE tsv @@ q ORDER BY score DESC LIMIT 20;
```

Then merge the two lists with **reciprocal rank fusion**: each chunk
scores `1 / (60 + rank)` in each list it appears in, and the scores are
added. Ranks, not raw scores — so the two engines do not have to agree on
what a "score" means. A chunk that both engines like rises to the top.

## Reranking

Search is fast because it compares the question with each chunk
separately. A **cross-encoder reranker** reads the question and a chunk
together, which is slower and much sharper. So: retrieve wide (20–30
candidates), then let the reranker keep the best 3–5.

## The prompt

Tell the model to answer only from the numbered sources, to cite them
like `[1]`, and — most important — to say so when the sources do not
contain the answer. Return the sources to the caller, so a person can
check the citation.

## Evaluation: the part that separates guessing from engineering

Build a **golden set**: 30 or more real questions, each with the chunk
(or the document and page) that holds the answer and a short reference
answer. Include a few questions the documents cannot answer. Then measure
two halves separately:

- **Retrieval.** **Hit rate@k**: in what share of questions is the right
  chunk in the top k? **MRR**: on average, how high is it? (1 for first
  place, ½ for second, ⅓ for third.)
- **Answers.** **Faithfulness**: is every claim supported by the
  retrieved chunks? **Correctness**: does it mean the same as the
  reference? An LLM judges both with a fixed prompt — and you check a
  sample of its verdicts yourself.

Run the evaluation after every change and keep a results table.

## When an answer is wrong

Look at the retrieved chunks first. **Right chunk missing?** It is a
retrieval problem: parsing, chunking or search. **Right chunk present?**
It is a generation problem: the prompt, too much context, the model. This
one rule saves days of tweaking the wrong thing.

## In production

**Changed documents.** Re-ingest a changed document by deleting its old
chunks by document id first; otherwise the old and new versions both
answer.

**Access rules.** Filter by metadata in the query itself. Never retrieve
what the user may not see — a chunk in the prompt is a chunk the model
can repeat.

**Caching.** Questions repeat. Cache embeddings of frequent questions,
and whole answers where the documents rarely change (Level 5's sticky
notes).

**A latency budget.** Give each stage a budget — embed, search, rerank,
generate — and log the time of each. When answers get slow, you will know
which stage to blame instead of guessing.

## What to remember

1. **Wrong answer? Read the retrieved chunks first.** Most hallucinations are retrieval failures.
2. **A chunk is one idea that makes sense alone.**
3. **Vectors for meaning, keywords for exact things, a reranker to choose.**
4. **"Not in the documents" is a feature.**
5. **Change one thing, run the golden set, keep or revert.**
